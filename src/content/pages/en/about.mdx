import { siteConfig } from "~/config/site";
import { getCollection } from "astro:content";
import { getAllAuthors, visiblePosts } from "~/lib/posts";
import { t } from "i18n:astro";

export const allPosts = visiblePosts(await getCollection("posts"));
export const authorEntries = await getCollection("authors");
export const authors = getAllAuthors(allPosts, authorEntries);

## What gets published here

Long-form notes on building software: the decision behind a change, what it cost, and what
would be done differently next time. Posts are written to be read once, carefully, rather
than skimmed.

## How it is written

Every post aims to be reproducible and honest about trade-offs. If something did not work,
that goes in too. Corrections are made in place, with the update date shown at the top of
the post.
